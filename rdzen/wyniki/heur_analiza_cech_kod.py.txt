class Rec:
    nazwa="rec"
    def __init__(s, sw, buf): s.sw, s.buf, s.h = sw, buf, HeurystykaOgolnosci()
    def ocen(s, K, ctx):
        kons, p = ctx.konsolidacja, ctx.pamiec; z=np.asarray(ctx.znane_magmy, dtype=np.int64)
        F=kons.macierz_zaleznosci; ri={r:i for i,r in enumerate(kons.reprezentanci)}
        size={}; 
        for a,r in kons.cls.items(): size[r]=size.get(r,0)+1
        W,c=s.sw.world,s.sw.concepts; n=max(ctx.liczba_znanych_magm,1)
        f=lambda x: float(p.concepts[x][z].mean()) if len(z) else .5
        for k in K:
            A,B=kons.cls[k.a],kons.cls[k.b]; i,j=ri[A],ri[B]
            t= not (c[k.a][W] & ~c[k.b][W]).any(); tr=(s.sw.cls[k.a],s.sw.cls[k.b]) in s.sw.TR
            s.buf.append(dict(smiale=k.zrodlo=="sen:smiale", sup=k.wsparcie/n, fa=f(k.a), fb=f(k.b), sa=size[A], sb=size[B],
              outa=int(F[i].sum()), ina=int(F[:,i].sum()), outb=int(F[j].sum()), inb=int(F[:,j].sum()),
              ta=typ_pojecia(A), tb=typ_pojecia(B), y_t=t, y_tr=t and tr, pula=len(K), sen=len(s.buf)))
        return s.h.ocen(K, ctx)

def jeden(a):
    H,rz=a; sw=zbuduj_swiat(H,rz,SLOWNIK); buf=[]
    for sd in range(1610,1614):
        b0=len(buf)

    return buf
